async function loadSupportVenues() {
  document.getElementById('support-venue-list').style.display='block';
  document.getElementById('support-chat').style.display='none';
  var el=document.getElementById('support-venue-list');
  el.innerHTML='<div style="text-align:center;padding:20px"><div style="animation:spin .7s linear infinite;display:inline-block;font-size:20px">&#9675;</div></div>';
  // Get venues that have sent messages
  var msgs=await sbGet('support_messages?select=venue_id,read,created_at&from_platform=eq.false&order=created_at.desc');
  var venues=await sbGet('venues?select=id,name,slug&order=name.asc');
  // Group by venue_id, get latest message time and unread count
  var venueMap={};
  msgs.forEach(function(m){
    if(!venueMap[m.venue_id])venueMap[m.venue_id]={count:0,hasUnread:false,latest:m.created_at};
    venueMap[m.venue_id].count++;
    if(!m.read)venueMap[m.venue_id].hasUnread=true;
  });
  var activeVenues=venues.filter(function(v){return venueMap[v.id];});
  if(!activeVenues.length){el.innerHTML='<div class="empty">Henuz destek talebi yok.</div>';return;}
  el.innerHTML='';
  activeVenues.forEach(function(v){
    var info=venueMap[v.id];
    var c=document.createElement('div');c.className='card';
    var h=document.createElement('div');h.className='ch';
    var nd=document.createElement('div');nd.style.flex='1';
    var nm=document.createElement('div');nm.className='cn';nm.textContent=v.name;
    var mt=document.createElement('div');mt.className='cm';mt.textContent=info.count+' mesaj';
    if(info.hasUnread){var badge=document.createElement('span');badge.style.cssText='background:#C0392B;color:#fff;font-family:Space Mono,monospace;font-size:7px;padding:2px 7px;border-radius:10px;margin-left:6px';badge.textContent='YENİ';nm.appendChild(badge);}
    nd.appendChild(nm);nd.appendChild(mt);h.appendChild(nd);
    h.onclick=(function(vid,vname){return function(){openSupportChat(vid,vname);};})(v.id,v.name);
    c.appendChild(h);el.appendChild(c);
  });
}

async function renderSupportMsgs(venueId) {
  var data=await sbGet('support_messages?venue_id=eq.'+venueId+'&order=created_at.asc&limit=50');
  var el=document.getElementById('support-msgs');if(!el)return;
  el.innerHTML='';
  data.forEach(function(m){
    var fromPlatform=m.from_platform;
    var d='';try{d=new Date(m.created_at).toLocaleTimeString('tr-TR',{hour:'2-digit',minute:'2-digit'});}catch(e){}
    var bubble=document.createElement('div');
    bubble.style.cssText='display:flex;margin-bottom:10px;'+(fromPlatform?'justify-content:flex-end':'justify-content:flex-start');
    var inner=document.createElement('div');
    inner.style.cssText='max-width:80%;padding:10px 13px;border-radius:10px;'
      +(fromPlatform?'background:#F0EDE8;color:#060606;border-bottom-right-radius:2px':'background:#1C1C1C;border-bottom-left-radius:2px');
    inner.innerHTML='<div style="font-size:13px;line-height:1.5;color:'+(fromPlatform?'#060606':'#F0EDE8')+'">'+esc(m.message)+'</div>'
      +'<div style="font-family:Space Mono,monospace;font-size:7px;color:'+(fromPlatform?'rgba(0,0,0,.4)':'#444')+';margin-top:4px;text-align:right">'+d+(fromPlatform?' · SEN':'')+'</div>';

  // Mark as read
  data.filter(function(m){return !m.from_platform&&!m.read;}).forEach(function(m){
    sbPatch('support_messages',m.id,{read:true});
  });
}

function sbGet(path){
  return fetch(SB+'/rest/v1/'+path,{headers:{'apikey':KEY,'Authorization':'Bearer '+KEY}})
    .then(function(r){return r.json();})
    .then(function(d){return Array.isArray(d)?d:[];})
    .catch(function(){return[];});
}

function sbPatch(table, id, data){
  return fetch(SB+'/rest/v1/'+table+'?id=eq.'+id,{
    method:'PATCH',
    headers:{'apikey':KEY,'Authorization':'Bearer '+KEY,'Content-Type':'application/json','Prefer':'return=representation'},
    body:JSON.stringify(data)
  }).then(function(r){return r.json();}).catch(function(){return null;});
}

  }).then(function(r){return r.json();}).catch(function(){return null;});
}
