  var slug=ge('cat-slug').value.trim().toLowerCase().replace(/\s+/g,'-');
  var order=parseInt(ge('cat-order').value)||0;
  if(!name||!slug){toast('Ad ve slug zorunlu');return;}
  var r=await sbPost('categories',{name:name,slug:slug,icon:icon,sort_order:order,active:true});
  if(r&&!r.code){
    ge('cat-icon').value='';ge('cat-name').value='';ge('cat-slug').value='';ge('cat-order').value='';
    toast(name+' eklendi');loadCats();
  } else {
    toast('Hata — slug zaten var olabilir');
  }
}

async function toggleCat(id, active){
  await sbPatch('categories',id,{active:active});
  toast(active?'Gösterildi':'Gizlendi');loadCats();
}

async function updateCat(id, data){
  await sbPatch('categories',id,data);
  loadCats();
}

async function deleteCat(id){
  await sbDelete('categories',id);
  toast('Silindi');loadCats();
}

async function loadUsers(){
  var [profiles, stats] = await Promise.all([
    sbGet('profiles?select=id,name,email,status,tier,trust_score,total_attended,created_at,venue_id&order=created_at.desc&limit=50'),
    sbGet('profiles?select=status')
  ]);
  var tot=stats.length, app=stats.filter(function(p){return p.status==='approved';}).length, pend=stats.filter(function(p){return p.status==='pending';}).length;
  ge('user-stats').innerHTML=
    '<div class="st"><div class="st-n g">'+app+'</div><div class="st-l">ONAYLANDI</div></div>'
    +'<div class="st"><div class="st-n go">'+pend+'</div><div class="st-l">BEKLEYEN</div></div>'
    +'<div class="st"><div class="st-n b">'+tot+'</div><div class="st-l">TOPLAM</div></div>';
  var el=ge('user-list');
  if(!profiles.length){el.innerHTML='<div class="empty">Kayıtlı kullanıcı yok.</div>';return;}
  el.innerHTML='';
  profiles.forEach(function(p){
    var bc={pending:'bp',approved:'ba',rejected:'br2',frozen:'bu'}[p.status]||'bu';
    var bt={pending:'BEKLEMEDE',approved:'ONAYLANDI',rejected:'REDDEDİLDİ',frozen:'DONDURULDU'}[p.status]||p.status;
    var card=document.createElement('div');card.className='card';
    var hdr=document.createElement('div');hdr.className='ch';
    hdr.onclick=function(){this.nextElementSibling.classList.toggle('op');};
    var nd=document.createElement('div');nd.style.flex='1';
    var nm=document.createElement('div');nm.className='cn';nm.textContent=p.name||'—';
    var mt=document.createElement('div');mt.className='cm';mt.textContent=(p.email||'—')+' · Puan: '+(p.trust_score||0)+' · Katılım: '+(p.total_attended||0);
    var bg=document.createElement('span');bg.className='badge '+bc;bg.textContent=bt;

    var body=document.createElement('div');body.className='cb';

    body.appendChild(det);card.appendChild(body);el.appendChild(card);
  });
}
