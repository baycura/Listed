function db(table) {
  var _t = table, _f = [], _sel = '*', _ord = null, _lim = null, _single = false;
  function url() {
    var u = SB + '/rest/v1/' + _t + '?select=' + _sel;
    _f.forEach(function(f) { u += '&' + f; });
    if (_ord) u += '&order=' + _ord;
    if (_lim) u += '&limit=' + _lim;
    return u;
  }
  function h(extra) {
    var hd = { 'apikey': KEY, 'Authorization': 'Bearer ' + KEY, 'Content-Type': 'application/json', 'Prefer': 'return=representation' };
    if (extra) Object.keys(extra).forEach(function(k) { hd[k] = extra[k]; });
    return hd;
  }
  var q = {
    select: function(s) { _sel = s; return q; },
    eq: function(c, v) { _f.push(c + '=eq.' + encodeURIComponent(v)); return q; },
    gte: function(c, v) { _f.push(c + '=gte.' + encodeURIComponent(v)); return q; },
    order: function(c, o) { _ord = c + (o && o.ascending === false ? '.desc' : '.asc'); return q; },
    limit: function(n) { _lim = n; return q; },
    single: function() { _single = true; return q; },
    insert: function(data) {
      return fetch(SB + '/rest/v1/' + _t, { method: 'POST', headers: h(), body: JSON.stringify(data) })

        .catch(function(e) { return { data: null, error: { message: e.message } }; });
    },
    update: function(data) {
      return fetch(url(), { method: 'PATCH', headers: h(), body: JSON.stringify(data) })
        .then(function(r) { return r.json().then(function(d) { return { data: d, error: r.ok ? null : { message: JSON.stringify(d) } }; }); })
        .catch(function(e) { return { data: null, error: { message: e.message } }; });
    },
    delete: function() {
      return fetch(url(), { method: 'DELETE', headers: h() })
        .then(function(r) { return { data: null, error: r.ok ? null : { message: 'err' } }; })
        .catch(function(e) { return { data: null, error: { message: e.message } }; });

    sessionStorage.removeItem('sub_adm_' + SLUG);
    show('s-admin'); aTab('ov'); return;
  }
  // 3. Check sub-admins in venue settings
  var subs = (venue && venue.settings && venue.settings.sub_admins) || [];
  var sub = subs.find(function(s){ return s.username === user && s.password === pw; });

    sessionStorage.removeItem('adm_' + SLUG);

function compress(file, cb) {
  var reader = new FileReader();
  reader.onload = function(e) {
    var img = new Image();
    img.onload = function() {
      var canvas = document.createElement('canvas');
      var maxW = 800, maxH = 1000, w = img.width, h = img.height;
      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
      if (h > maxH) { w = Math.round(w * maxH / h); h = maxH; }
      canvas.width = w; canvas.height = h;
      canvas.getContext('2d').drawImage(img, 0, 0, w, h);
      // Try quality 0.75, check size, lower if needed
      var b64 = canvas.toDataURL('image/jpeg', 0.75);
      var kb = Math.round(b64.length * 0.75 / 1024);
      if (kb > 300) b64 = canvas.toDataURL('image/jpeg', 0.55);
      if (kb > 500) b64 = canvas.toDataURL('image/jpeg', 0.4);
      kb = Math.round(b64.length * 0.75 / 1024);
      cb(b64, kb);
    };
    img.src = e.target.result;
  };
  reader.readAsDataURL(file);
}

async function translateText(text, from, to){
  try{
    var url='https://api.mymemory.translated.net/get?q='+encodeURIComponent(text)+'&langpair='+from+'|'+to;
    var r=await fetch(url,{signal:AbortSignal.timeout(5000)});
    var d=await r.json();
    if(d.responseStatus===200&&d.responseData&&d.responseData.translatedText){
      return d.responseData.translatedText;
    }
  }catch(e){}
  return null;
}

async function translateEvtFields(){

async function apprvR(id, evId, cnt, cid) {

  if (evId) {
    var ev = await db('events').select('approved_count').eq('id', evId).single();

function sendWA(r) {

  var dateStr = '';
  try {
    var d = new Date((r.event_date||'')+'T12:00:00');
    var days = ['Pazar','Pazartesi','Sali','Carsamba','Persembe','Cuma','Cumartesi'];
    var months = ['Ocak','Subat','Mart','Nisan','Mayis','Haziran','Temmuz','Agustos','Eylul','Ekim','Kasim','Aralik'];
    dateStr = days[d.getDay()]+', '+d.getDate()+' '+months[d.getMonth()];
  } catch(e) { dateStr = r.event_date||''; }
  var venueName = venue ? venue.name : 'Listed';
  var lines = [
    '✅ *Rezervasyonunuz Onaylandı*',
    '',
    '*' + (r.event_name||'Etkinlik') + '*',
    '📅 ' + dateStr + (r.event_time ? ' · ' + r.event_time : ''),
    '👥 ' + (r.guest_count||1) + ' kişi',
    (r.table_name ? '🪑 Masa: ' + r.table_name : ''),
    '',
    '🎫 *Giriş Kodunuz:*',
    '*' + (r.qr_id||'—') + '*',
    '',
    'Kapıda bu kodu göstermeniz yeterlidir.',
    '',
    '— ' + venueName
  ].filter(function(l){ return l !== null && !(l==='' && lines && lines[lines.length-1]===''); });
  var msg = lines.map(function(l){ return encodeURIComponent(l); }).join('%0A');
  window.open('https://wa.me/' + ph + '?text=' + msg, '_blank');
}

  var res = document.getElementById('qr-res');
  if (!inp) { toast('QR kodu girin'); return; }

  function showRes(bg, border, color, html) {
    res.style.cssText = 'display:block;border-radius:2px;padding:20px 16px;font-family:Space Mono,monospace;font-size:10px;letter-spacing:1px;line-height:2;text-align:center;background:'+bg+';color:'+color+';border:1px solid '+border;
    res.innerHTML = html;
  }

  res.style.display = 'block';
  showRes('rgba(100,100,100,.06)','#2A2A2A','#666','Sorgulanıyor...');

  var r = await db('reservations').select('*').eq('qr_id', inp).single();
  var rv = r.data;

  if (!rv) {
    showRes('rgba(192,57,43,.06)','rgba(192,57,43,.3)','#C0392B',
      '<div style="font-size:32px;font-weight:900;letter-spacing:4px;margin-bottom:8px">GEÇERSİZ</div>'
      +'<div style="font-size:8px;color:#C0392B">Kod bulunamadı</div>');
    return;
  }

  if (rv.status === 'used') {
    showRes('rgba(100,100,100,.06)','#2A2A2A','#666',
      '<div style="font-size:28px;font-weight:900;letter-spacing:3px;margin-bottom:10px">KULLANILMIŞ</div>'
      +'<div style="font-size:11px;color:#888;margin-bottom:6px">'+esc(rv.name)+'</div>'
      +'<div style="font-size:8px;color:#444">Bu bilet daha önce kullanıldı.<br>Giriş verilemiyor.</div>'
      +(rv.checked_in_at ? '<div style="font-size:7px;color:#333;margin-top:6px">Kullanım: '+new Date(rv.checked_in_at).toLocaleTimeString('tr-TR')+'</div>' : ''));
    return;
  }

  if (rv.status !== 'approved') {
    showRes('rgba(192,57,43,.06)','rgba(192,57,43,.3)','#C0392B',
      '<div style="font-size:28px;font-weight:900;letter-spacing:3px;margin-bottom:10px">ONAYSIZ</div>'

      +'<div style="font-size:8px;color:#C0392B;margin-top:4px">Durum: '+rv.status+'</div>');

  showRes('rgba(201,168,76,.06)','rgba(201,168,76,.3)','#C9A84C',
    '<div style="font-size:10px;letter-spacing:2px;margin-bottom:12px">KİMLİK DOĞRULAMA</div>'
    +'<div style="font-size:22px;font-weight:900;letter-spacing:2px;color:#F0EDE8;margin-bottom:4px">'+esc(rv.name)+'</div>'
    +'<div style="font-size:9px;color:#888;margin-bottom:12px">'+esc(rv.event_name||'')+(rv.event_time?' · '+rv.event_time:'')+'</div>'
    +'<div style="font-size:11px;color:#C9A84C;margin-bottom:'+(rv.table_name?'4px':'16px')+'">'+( rv.guest_count||1)+' kişi</div>'
    +(rv.table_name?'<div style="font-size:9px;color:#888;margin-bottom:16px">Masa: '+esc(rv.table_name)+'</div>':'')

async function confirmEntry() {
  var resId = window._confirmId;
  var res = document.getElementById('qr-res');
  res.innerHTML = '<div style="animation:spin .7s linear infinite;display:inline-block;font-size:20px">◌</div>';
  // Get reservation details
  var rv = await db('reservations').select('*').eq('id', resId).single();
  var r = rv.data;
  await db('reservations').eq('id', resId).update({ status: 'used', checked_in_at: new Date().toISOString() });
  // Award points to the attendee
  if(r && r.profile_id){
    var prof = await db('profiles').select('trust_score,tier,total_attended,attendance_streak,referred_by_id,first_attended').eq('id', r.profile_id).single();
    var p = prof.data;
    if(p){
      var isFirst = !p.first_attended;
      var newScore = (p.trust_score||0) + 10;
      var newAttended = (p.total_attended||0) + 1;
      var newStreak = (p.attendance_streak||0) + 1;
      // Streak bonus
      if(newStreak % 3 === 0) newScore += 50;
      var newTier = newScore>=1500?'elite':newScore>=600?'gold':newScore>=200?'silver':'bronze';
      var upd = {trust_score:newScore, tier:newTier, total_attended:newAttended, attendance_streak:newStreak};
      if(isFirst) upd.first_attended = new Date().toISOString();
      await db('profiles').eq('id', r.profile_id).update(upd);
      await db('reputation_ledger').insert({profile_id:r.profile_id,score_type:'trust',delta:10,balance_after:newScore,reason_code:'checkin',source_type:'event',venue_id:venue?venue.id:null}).catch(function(){});
      // Referral +100 on first attendance
      if(isFirst && p.referred_by_id){
        var refP = await db('profiles').select('trust_score,tier').eq('id', p.referred_by_id).single();
        if(refP.data){
          var rs = (refP.data.trust_score||0)+100;
          var rt = rs>=1500?'elite':rs>=600?'gold':rs>=200?'silver':'bronze';
          await db('profiles').eq('id', p.referred_by_id).update({trust_score:rs, tier:rt});
          await db('reputation_ledger').insert({profile_id:p.referred_by_id,score_type:'trust',delta:100,balance_after:rs,reason_code:'referral_attended',source_type:'referral',venue_id:venue?venue.id:null}).catch(function(){});
        }
      }
    }
  }
  res.style.cssText = 'display:block;border-radius:2px;padding:20px 16px;font-family:Space Mono,monospace;font-size:10px;letter-spacing:1px;line-height:2;text-align:center;background:rgba(61,155,90,.08);color:#3D9B5A;border:1px solid rgba(61,155,90,.3)';
  res.innerHTML = '<div style="font-size:40px;margin-bottom:8px">&#10035;</div><div style="font-size:28px;font-weight:900;letter-spacing:4px;margin-bottom:8px">GİRİŞ OK</div><div style="font-size:8px;color:#3D9B5A;letter-spacing:1px">+10 puan · Kayit islendi</div>';
}

function onLogo(ev) {
  var f = ev.target.files[0]; if (!f) return;
  compress(f, function(b64, kb) {

function addTable() {
  var inp = document.getElementById('masa-inp'); if (!inp) return;
  var nm = inp.value.trim(); if (!nm) { toast('Masa adi bos olamaz'); return; }
  var cap = parseInt(document.getElementById('masa-cap').value) || 0;
  var min = parseInt(document.getElementById('masa-min').value) || 0;
  var zone = document.getElementById('masa-zone').value || '';
  if (!venue.settings) venue.settings = {};
  if (!venue.settings.tables_data) venue.settings.tables_data = [];
  if (!venue.settings.tables) venue.settings.tables = [];
  var exists = venue.settings.tables_data.find(function(t){ return t.name === nm; });
  if (exists) { toast('Bu isimde masa zaten var'); return; }
  var tableObj = { name: nm, capacity: cap, min_spend: min, zone: zone, status: 'active' };
  venue.settings.tables_data.push(tableObj);
  venue.settings.tables = venue.settings.tables_data.map(function(t){ return t.name; });
  inp.value = '';
  document.getElementById('masa-cap').value = '';
  document.getElementById('masa-min').value = '';
  document.getElementById('masa-zone').value = '';
  renderTablesAdmin(venue.settings.tables_data);
  saveTables();
  toast(nm + ' eklendi');
}

function removeTable(nm) {
  if (!venue || !venue.settings) return;
  if (venue.settings.tables_data) {
    venue.settings.tables_data = venue.settings.tables_data.filter(function(t){ return t.name !== nm; });
    venue.settings.tables = venue.settings.tables_data.map(function(t){ return t.name; });
    renderTablesAdmin(venue.settings.tables_data);
  } else if (venue.settings.tables) {
    venue.settings.tables = venue.settings.tables.filter(function(t){ return t !== nm; });
    renderTablesAdmin(venue.settings.tables);
  }
  saveTables();
}

function renderTablesAdmin(tables) {
  var el = document.getElementById('masa-list'); if (!el) return;
  if (!tables || !tables.length) {
    el.innerHTML = '<div style="font-family:Space Mono,monospace;font-size:8px;color:#444;letter-spacing:1px;padding:12px 0;text-align:center">Henuz masa eklenmedi</div>';
    return;
  }
  el.innerHTML = '';
  var zoneLabels = { ic:'İc Alan', dis:'Dis Alan', teras:'Teras', vip:'VIP', bar:'Bar Onü', sahne:'Sahne Onü' };
  tables.forEach(function(t) {
    var nm = typeof t === 'string' ? t : t.name;
    var cap = typeof t === 'object' ? t.capacity : 0;
    var minS = typeof t === 'object' ? t.min_spend : 0;
    var zone = typeof t === 'object' ? t.zone : '';
    var card = document.createElement('div');
    card.style.cssText = 'background:#0a0a0a;border:1px solid #1C1C1C;border-radius:2px;margin-bottom:6px;overflow:hidden';
    var top = document.createElement('div');
    top.style.cssText = 'display:flex;align-items:center;justify-content:space-between;padding:10px 12px';
    var info = document.createElement('div');
    info.style.flex = '1';
    var nameEl = document.createElement('div');
    nameEl.style.cssText = 'font-family:Space Mono,monospace;font-size:13px;font-weight:700;color:#F0EDE8;letter-spacing:1px';
    nameEl.textContent = nm;
    var meta = document.createElement('div');
    meta.style.cssText = 'font-family:Space Mono,monospace;font-size:7px;color:#666;letter-spacing:1px;margin-top:3px';
    var metaParts = [];
    if (cap) metaParts.push(cap + ' kisi');
    if (minS) metaParts.push('min ₺'+minS);
    if (zone && zoneLabels[zone]) metaParts.push(zoneLabels[zone]);
    meta.textContent = metaParts.join(' · ') || 'Detay girilmedi';
    info.appendChild(nameEl); info.appendChild(meta);
    var delBtn = document.createElement('button');
    delBtn.style.cssText = 'background:rgba(192,57,43,.1);border:1px solid rgba(192,57,43,.2);color:#C0392B;border-radius:2px;padding:6px 10px;font-size:12px;cursor:pointer;-webkit-appearance:none';
    delBtn.textContent = 'SİL';
    (function(tn){ delBtn.onclick = function(){ if(confirm(tn+' silinsin mi?')) removeTable(tn); }; })(nm);
    top.appendChild(info); top.appendChild(delBtn);
    card.appendChild(top);
    el.appendChild(card);
  });
  // Summary
  var sum = document.createElement('div');
  sum.style.cssText = 'font-family:Space Mono,monospace;font-size:8px;color:#3D9B5A;letter-spacing:1px;padding:8px 0;text-align:center';
  var totalCap = tables.reduce(function(acc,t){ return acc+(typeof t==='object'?t.capacity||0:0); },0);
  sum.textContent = tables.length + ' masa' + (totalCap?' · toplam '+totalCap+' kisi':'');
  el.appendChild(sum);
}

var _cameraStream = null;
var _cameraTimer = null;

async function startCamera(){
  var wrap = document.getElementById('camera-wrap');
  var video = document.getElementById('camera-video');
  if(!wrap || !video){ toast('Kamera desteklenmiyor'); return; }
  if(!navigator.mediaDevices||!navigator.mediaDevices.getUserMedia){ toast('Tarayici kamera desteklemiyor'); return; }
  try{
    _cameraStream = await navigator.mediaDevices.getUserMedia({video:{facingMode:'environment'},audio:false});
    video.srcObject = _cameraStream;
    wrap.style.display = 'block';
    // Use BarcodeDetector if available
    if('BarcodeDetector' in window){
      var detector = new BarcodeDetector({formats:['qr_code']});
      _cameraTimer = setInterval(async function(){
        if(video.readyState !== 4) return;
        try{
          var barcodes = await detector.detect(video);
          if(barcodes.length){
            var code = barcodes[0].rawValue;
            stopCamera();
            document.getElementById('qr-inp').value = code.toUpperCase();
            scanQR();
          }
        }catch(e){}
      }, 500);
    } else {
      // Fallback: manual entry with camera as viewfinder
      toast('Kamera acildi. QR kodu manuel girin.');
    }
  }catch(e){
    toast('Kamera izni verilmedi: '+e.message);
  }
}

function stopCamera(){
  if(_cameraTimer){ clearInterval(_cameraTimer); _cameraTimer=null; }
  if(_cameraStream){ _cameraStream.getTracks().forEach(function(t){t.stop();}); _cameraStream=null; }
  var wrap = document.getElementById('camera-wrap');
  if(wrap) wrap.style.display = 'none';
}

}

async function checkAutoBlacklist(profileId) {
  var r=await db('warnings').select('level').eq('profile_id',profileId).eq('active',true);
  var warnings=r.data||[];
  var reds=warnings.filter(function(w){return w.level===3;}).length;
  if(reds>=3||warnings.length>=5){
    await db('profiles').eq('id',profileId).update({status:'frozen',trust_score:0});
    toast('OTOMATIK KARA LISTE: '+warnings.length+' aktif uyari');
    return true;
  }
  return false;
}

async function exportCSV() {
  var evSel=document.getElementById('res-ev-filter');
  var evId=evSel&&evSel.value;
  var q=db('reservations').select('name,phone,guest_count,status,qr_id,event_name,event_date,event_time,table_name,checked_in_at').eq('venue_id',venue.id);

  a.click();URL.revokeObjectURL(url);
  toast('CSV indirildi: '+data.length+' kayit');

// ── INVITES ────────────────────────────────────────────────────
var INVITE_LIMIT = 100;

async function loadInvites() {
  if (!venue) return;
  var r = await db('settings').select('key,value').eq('venue_id', venue.id).eq('key', 'invites').single();
  var invites = [];
  try { if(r.data) invites = JSON.parse(r.data.value)||[]; } catch(e){}
  var statsEl = document.getElementById('invite-stats');
  var listEl = document.getElementById('invite-list');
  if(statsEl) statsEl.textContent = invites.length + ' / ' + INVITE_LIMIT + ' davet oluşturuldu';
  if(!listEl) return;
  if(!invites.length){ listEl.innerHTML='<div style="font-family:Space Mono,monospace;font-size:8px;color:#444;text-align:center;padding:12px">Henüz davet yok.</div>'; return; }
  listEl.innerHTML='';
  invites.slice(-10).reverse().forEach(function(inv){
    var url=location.origin+location.pathname.replace('admin.html','')+'?v='+venue.slug+'&invite='+inv.code;
    var d='';try{d=new Date(inv.created_at).toLocaleDateString('tr-TR',{day:'numeric',month:'short'});}catch(e){}
    var row=document.createElement('div');
    row.style.cssText='background:#0a0a0a;border:1px solid #1C1C1C;border-radius:2px;padding:10px 12px;margin-bottom:6px;display:flex;align-items:center;gap:8px';
    row.innerHTML='<div style="flex:1;min-width:0">'
      +'<div style="font-family:Space Mono,monospace;font-size:9px;color:#C9A84C;letter-spacing:2px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap">'+inv.code+'</div>'
      +'<div style="font-family:Space Mono,monospace;font-size:7px;color:#444;margin-top:2px">'+d+(inv.used?' · KULLANILDI':' · Aktif')+'</div>'
      +'</div>'
      +'<button data-url="'+url+'" class="copy-invite-btn" style="background:none;border:1px solid #2A2A2A;color:#666;border-radius:2px;padding:6px 10px;font-family:Space Mono,monospace;font-size:8px;cursor:pointer;-webkit-appearance:none;flex-shrink:0">KOPYALA</button>';
    listEl.appendChild(row);
  });
}

document.addEventListener('click', function(e){
  if(e.target.classList.contains('copy-invite-btn')){
    var url=e.target.getAttribute('data-url');
    if(url){
      if(navigator.clipboard){navigator.clipboard.writeText(url).then(function(){toast('Davet linki kopyalandı');});}
      else toast(url);
    }
  }
});

async function createInvite() {
  if (!venue) return;
  var r = await db('settings').select('key,value').eq('venue_id',venue.id).eq('key','invites').single();
  var invites = [];
  try { if(r.data) invites = JSON.parse(r.data.value)||[]; } catch(e){}
  if(invites.length >= INVITE_LIMIT){ toast('Davet limitine ulaşıldı ('+INVITE_LIMIT+')'); return; }
  var code = 'INV-' + rndHex(8).toUpperCase();
  invites.push({code:code,created_at:new Date().toISOString(),used:false});
  var upsertData = {key:'invites',value:JSON.stringify(invites),venue_id:venue.id};
  await db('settings').upsert(upsertData);
  toast('Davet kodu oluşturuldu: '+code);
  loadInvites();
}

// ── EVENT RATINGS ─────────────────────────────────────────────
async function loadEventRatings(evId) {
  var el = document.getElementById('erating-'+evId); if(!el) return;
  var r = await db('event_ratings').select('rating').eq('event_id', evId).catch(function(){return{data:[]};});
  var data = r.data||[];
  if(!data.length){ el.textContent = 'Henuz puan verilmedi'; return; }
  var avg = data.reduce(function(s,d){return s+(d.rating||0);},0)/data.length;
  var stars = '';
  for(var i=1;i<=5;i++) stars += i<=Math.round(avg)?'★':'☆';
  el.innerHTML = '<span style="color:#C9A84C;font-size:14px">'+stars+'</span>'
    +' <span style="color:#C9A84C">'+avg.toFixed(1)+'</span>'
    +' <span style="color:#444">('+data.length+' oy)</span>';
}

// ── 24H REMINDER ──────────────────────────────────────────────
async function sendReminders(evId, evName) {
  var r = await db('reservations').select('name,phone,guest_count,qr_id').eq('event_id', evId).eq('status', 'approved');
  var list = r.data||[];
  if(!list.length){ toast('Onaylı rezervasyon yok'); return; }
  if(!confirm(list.length+' kisiye hatirlama gonderilsin mi?')){ return; }
  var venueName = venue ? venue.name : 'Listed';
  list.forEach(function(rv, i){
    setTimeout(function(){
      var ph = (rv.phone||'').replace(/[^0-9]/g,'');
      if(!ph) return;
      var lines = [
        'Etkinlik Hatirlatmasi',
        '*'+evName+'* yarin!',
        (rv.guest_count||1)+' kisi',
        'QR: '+(rv.qr_id||''),
        'Gorusuruz — '+venueName
      ];
      var msg = encodeURIComponent(lines.join('%0A'));
      window.open('https://wa.me/'+ph+'?text='+msg,'_blank');
    }, i * 2000);
  });
  toast('Hatirlatmalar gonderiliyor: '+list.length+' kisi');
}
