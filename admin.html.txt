function db(table) {
  var _t = table, _f = [], _sel = '*', _ord = null, _lim = null, _single = false;
  function url() {
    var u = SB + '/rest/v1/' + _t + '?select=' + _sel;
    _f.forEach(function(f) { u += '&' + f; });
    if (_ord) u += '&order=' + _ord;
    if (_lim) u += '&limit=' + _lim;
    return u;
  }
  function h(extra) {
    var hd = { 'apikey': KEY, 'Authorization': 'Bearer ' + KEY, 'Content-Type': 'application/json', 'Prefer': 'return=representation' };
    if (extra) Object.keys(extra).forEach(function(k) { hd[k] = extra[k]; });
    return hd;
  }
  var q = {
    select: function(s) { _sel = s; return q; },
    eq: function(c, v) { _f.push(c + '=eq.' + encodeURIComponent(v)); return q; },
    gte: function(c, v) { _f.push(c + '=gte.' + encodeURIComponent(v)); return q; },
    order: function(c, o) { _ord = c + (o && o.ascending === false ? '.desc' : '.asc'); return q; },
    limit: function(n) { _lim = n; return q; },
    single: function() { _single = true; return q; },
    insert: function(data) {
      return fetch(SB + '/rest/v1/' + _t, { method: 'POST', headers: h(), body: JSON.stringify(data) })
        .then(function(r) { return r.json().then(function(d) { return { data: Array.isArray(d) ? d[0] : d, error: r.ok ? null : { message: JSON.stringify(d), code: d && d.code } }; }); })
        .catch(function(e) { return { data: null, error: { message: e.message } }; });
    },
    update: function(data) {
      return fetch(url(), { method: 'PATCH', headers: h(), body: JSON.stringify(data) })
        .then(function(r) { return r.json().then(function(d) { return { data: d, error: r.ok ? null : { message: JSON.stringify(d) } }; }); })
        .catch(function(e) { return { data: null, error: { message: e.message } }; });
    },
    delete: function() {
      return fetch(url(), { method: 'DELETE', headers: h() })
        .then(function(r) { return { data: null, error: r.ok ? null : { message: 'err' } }; })
        .catch(function(e) { return { data: null, error: { message: e.message } }; });
    },
    then: function(res, rej) {
      var u = url();
      if (_single) u += '&limit=1';
      return fetch(u, { headers: h() })
        .then(function(r) {
          return r.json().then(function(d) {

            return result;
          });
        })
        .catch(function(e) { return { data: null, error: { message: e.message } }; })
        .then(res, rej);
    }
  };
  return q;
}

function compress(file, cb) {
  var reader = new FileReader();
  reader.onload = function(e) {
    var img = new Image();
    img.onload = function() {
      var canvas = document.createElement('canvas');
      var maxW = 800, maxH = 1000, w = img.width, h = img.height;
      if (w > maxW) { h = Math.round(h * maxW / w); w = maxW; }
      if (h > maxH) { w = Math.round(w * maxH / h); h = maxH; }
      canvas.width = w; canvas.height = h;
      canvas.getContext('2d').drawImage(img, 0, 0, w, h);
      // Try quality 0.75, check size, lower if needed
      var b64 = canvas.toDataURL('image/jpeg', 0.75);
      var kb = Math.round(b64.length * 0.75 / 1024);
      if (kb > 300) b64 = canvas.toDataURL('image/jpeg', 0.55);
      if (kb > 500) b64 = canvas.toDataURL('image/jpeg', 0.4);
      kb = Math.round(b64.length * 0.75 / 1024);
      cb(b64, kb);
    };
    img.src = e.target.result;
  };
  reader.readAsDataURL(file);
}

      var b2 = document.createElement('button'); b2.className = 'act rj'; b2.textContent = 'REDDET';
      (function(id) { b2.onclick = function() { rejR(id, cid); }; })(r.id);
      body.appendChild(b1); body.appendChild(b2);
    }
    if (r.status === 'approved' && r.phone) {
      var wa = document.createElement('button');
      wa.className = 'act ap'; wa.style.cssText = 'background:rgba(37,211,102,.08);color:#25d366;border-color:rgba(37,211,102,.2)';
      wa.textContent = 'WhatsApp';
      (function(rv) { wa.onclick = function() { sendWA(rv); }; })(r);
      body.appendChild(wa);
    }
    var det = document.createElement('div');
    det.style.cssText = 'font-family:Space Mono,monospace;font-size:8px;color:#666;line-height:2;padding:8px 0;border-top:1px solid #1C1C1C';
    det.textContent = 'Tel: ' + (r.phone || '--') + ' | QR: ' + (r.qr_id || '--') + ' | ' + fmtDate(r.event_date || '');
    body.appendChild(det); card.appendChild(body); el.appendChild(card);
  });
}

async function apprvR(id, evId, cnt, cid) {
  var res = await db('reservations').select('*').eq('id', id).single();
  var r = res.data;
  await db('reservations').eq('id', id).update({ status: 'approved', approved_at: new Date().toISOString() });
  if (evId) {
    var ev = await db('events').select('approved_count').eq('id', evId).single();
    if (ev.data) await db('events').eq('id', evId).update({ approved_count: (ev.data.approved_count || 0) + cnt });
  }
  toast('Onaylandi — WhatsApp aciliyor...');
  if (cid === 'rec-r') loadOv(); else loadRes();
  // Auto-open WhatsApp with ticket
  if (r) setTimeout(function(){ sendWA(r); }, 800);
}

async function rejR(id, cid) {
  await db('reservations').eq('id', id).update({ status: 'rejected' });
  toast('Reddedildi');
  if (cid === 'rec-r') loadOv(); else loadRes();
}

function sendWA(r) {
  var ph = (r.phone || '').replace(/[^0-9]/g, '');
  if (!ph) { toast('Telefon numarasi yok'); return; }
  // Format date nicely
  var dateStr = '';
  try {
    var d = new Date((r.event_date||'')+'T12:00:00');
    var days = ['Pazar','Pazartesi','Sali','Carsamba','Persembe','Cuma','Cumartesi'];
    var months = ['Ocak','Subat','Mart','Nisan','Mayis','Haziran','Temmuz','Agustos','Eylul','Ekim','Kasim','Aralik'];
    dateStr = days[d.getDay()]+', '+d.getDate()+' '+months[d.getMonth()];
  } catch(e) { dateStr = r.event_date||''; }
  var venueName = venue ? venue.name : 'Listed';
  var lines = [
    '✅ *Rezervasyonunuz Onaylandı*',
    '',
    '*' + (r.event_name||'Etkinlik') + '*',
    '📅 ' + dateStr + (r.event_time ? ' · ' + r.event_time : ''),
    '👥 ' + (r.guest_count||1) + ' kişi',
    (r.table_name ? '🪑 Masa: ' + r.table_name : ''),
    '',
    '🎫 *Giriş Kodunuz:*',
    '*' + (r.qr_id||'—') + '*',
    '',
    'Kapıda bu kodu göstermeniz yeterlidir.',
    '',
    '— ' + venueName
  ].filter(function(l){ return l !== null && !(l==='' && lines && lines[lines.length-1]===''); });
  var msg = lines.map(function(l){ return encodeURIComponent(l); }).join('%0A');
  window.open('https://wa.me/' + ph + '?text=' + msg, '_blank');
}

function addTable() {
  var inp = document.getElementById('masa-inp'); if (!inp) return;
  var nm = inp.value.trim(); if (!nm) { toast('Masa adi bos olamaz'); return; }
  var cap = parseInt(document.getElementById('masa-cap').value) || 0;
  var min = parseInt(document.getElementById('masa-min').value) || 0;
  var zone = document.getElementById('masa-zone').value || '';
  if (!venue.settings) venue.settings = {};
  if (!venue.settings.tables_data) venue.settings.tables_data = [];
  if (!venue.settings.tables) venue.settings.tables = [];
  var exists = venue.settings.tables_data.find(function(t){ return t.name === nm; });
  if (exists) { toast('Bu isimde masa zaten var'); return; }
  var tableObj = { name: nm, capacity: cap, min_spend: min, zone: zone, status: 'active' };
  venue.settings.tables_data.push(tableObj);
  venue.settings.tables = venue.settings.tables_data.map(function(t){ return t.name; });
  inp.value = '';
  document.getElementById('masa-cap').value = '';
  document.getElementById('masa-min').value = '';
  document.getElementById('masa-zone').value = '';
  renderTablesAdmin(venue.settings.tables_data);
  saveTables();
  toast(nm + ' eklendi');
}

function removeTable(nm) {
  if (!venue || !venue.settings) return;
  if (venue.settings.tables_data) {
    venue.settings.tables_data = venue.settings.tables_data.filter(function(t){ return t.name !== nm; });
    venue.settings.tables = venue.settings.tables_data.map(function(t){ return t.name; });
    renderTablesAdmin(venue.settings.tables_data);
  } else if (venue.settings.tables) {
    venue.settings.tables = venue.settings.tables.filter(function(t){ return t !== nm; });
    renderTablesAdmin(venue.settings.tables);
  }
  saveTables();
}

function renderTablesAdmin(tables) {
  var el = document.getElementById('masa-list'); if (!el) return;
  if (!tables || !tables.length) {
    el.innerHTML = '<div style="font-family:Space Mono,monospace;font-size:8px;color:#444;letter-spacing:1px;padding:12px 0;text-align:center">Henuz masa eklenmedi</div>';
    return;
  }
  el.innerHTML = '';
  var zoneLabels = { ic:'İc Alan', dis:'Dis Alan', teras:'Teras', vip:'VIP', bar:'Bar Onü', sahne:'Sahne Onü' };
  tables.forEach(function(t) {
    var nm = typeof t === 'string' ? t : t.name;
    var cap = typeof t === 'object' ? t.capacity : 0;
    var minS = typeof t === 'object' ? t.min_spend : 0;
    var zone = typeof t === 'object' ? t.zone : '';
    var card = document.createElement('div');
    card.style.cssText = 'background:#0a0a0a;border:1px solid #1C1C1C;border-radius:2px;margin-bottom:6px;overflow:hidden';
    var top = document.createElement('div');
    top.style.cssText = 'display:flex;align-items:center;justify-content:space-between;padding:10px 12px';
    var info = document.createElement('div');
    info.style.flex = '1';
    var nameEl = document.createElement('div');
    nameEl.style.cssText = 'font-family:Space Mono,monospace;font-size:13px;font-weight:700;color:#F0EDE8;letter-spacing:1px';
    nameEl.textContent = nm;
    var meta = document.createElement('div');
    meta.style.cssText = 'font-family:Space Mono,monospace;font-size:7px;color:#666;letter-spacing:1px;margin-top:3px';
    var metaParts = [];
    if (cap) metaParts.push(cap + ' kisi');
    if (minS) metaParts.push('min ₺'+minS);
    if (zone && zoneLabels[zone]) metaParts.push(zoneLabels[zone]);
    meta.textContent = metaParts.join(' · ') || 'Detay girilmedi';
    info.appendChild(nameEl); info.appendChild(meta);
    var delBtn = document.createElement('button');
    delBtn.style.cssText = 'background:rgba(192,57,43,.1);border:1px solid rgba(192,57,43,.2);color:#C0392B;border-radius:2px;padding:6px 10px;font-size:12px;cursor:pointer;-webkit-appearance:none';
    delBtn.textContent = 'SİL';
    (function(tn){ delBtn.onclick = function(){ if(confirm(tn+' silinsin mi?')) removeTable(tn); }; })(nm);
    top.appendChild(info); top.appendChild(delBtn);
    card.appendChild(top);
    el.appendChild(card);
  });
  // Summary
  var sum = document.createElement('div');
  sum.style.cssText = 'font-family:Space Mono,monospace;font-size:8px;color:#3D9B5A;letter-spacing:1px;padding:8px 0;text-align:center';
  var totalCap = tables.reduce(function(acc,t){ return acc+(typeof t==='object'?t.capacity||0:0); },0);
  sum.textContent = tables.length + ' masa' + (totalCap?' · toplam '+totalCap+' kisi':'');
  el.appendChild(sum);
}
