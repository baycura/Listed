function isThisWeekend(s){
  var d=new Date(s+'T12:00:00');
  var day=d.getDay();
  var today=new Date();today.setHours(0,0,0,0);
  var diff=Math.round((d-today)/864e5);

}

// ── OG TAGS ───────────────────────────────────────────────────────
function updateOG(title, desc, img, url){
  var tags={
    'og:title':title,'og:description':desc,'og:image':img||'','og:url':url||location.href,
    'twitter:title':title,'twitter:description':desc,'twitter:image':img||''
  };
  Object.keys(tags).forEach(function(k){
    var el=document.querySelector('meta[property="'+k+'"],meta[name="'+k+'"]');
    if(el)el.setAttribute('content',tags[k]);
  });
  document.title=title;
}

// ── WEB SHARE ─────────────────────────────────────────────────────
function shareEvent(e,vInfo){
  var vname=vInfo?vInfo.name:'';
  var url=location.origin+location.pathname+'?v='+(SLUG||(vInfo&&vInfo.slug)||'');
  var parts=[e.name];
  if(vname)parts.push('@ '+vname);
  if(e.date)parts.push(fmtDate(e.date));
  if(e.time)parts.push(e.time);
  var text=parts.join(' - ');
  if(navigator.share){
    navigator.share({title:e.name,text:text,url:url}).catch(function(){});
  } else {
    if(navigator.clipboard){navigator.clipboard.writeText(url).then(function(){toast('Link copied!');});}
    else toast(url);
  }
}

function compressImg(file,cb){
  var reader=new FileReader();
  reader.onload=function(e){
    var img=new Image(); img.onload=function(){
      var canvas=document.createElement('canvas'); var size=200;
      var w=img.width,h=img.height,min=Math.min(w,h);
      var sx=(w-min)/2,sy=(h-min)/2;
      canvas.width=size;canvas.height=size;
      canvas.getContext('2d').drawImage(img,sx,sy,min,min,0,0,size,size);
      cb(canvas.toDataURL('image/jpeg',.72));
    };img.src=e.target.result;
  };reader.readAsDataURL(file);
}
